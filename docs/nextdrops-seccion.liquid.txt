<script>
  (function () {
    var box = document.querySelector('[data-kc-countdown]');
    if (!box) return;
    var target = Date.parse(box.getAttribute('data-kc-countdown'));
    if (isNaN(target)) { box.hidden = true; return; }
    var u = {}; ['d', 'h', 'm', 's'].forEach(function (k) { u[k] = box.querySelector('[data-u="' + k + '"]'); });
    function pad(n) { return n < 10 ? '0' + n : '' + n; }
    function tick() {
      var diff = target - Date.now();
      if (diff <= 0) { box.innerHTML = '<p style="margin:0;letter-spacing:.2em;text-transform:uppercase;font-size:.8rem">Ya es la hora: actualiza la página para ver el lanzamiento</p>'; return; }
      var s = Math.floor(diff / 1000);
      u.d.textContent = Math.floor(s / 86400);
      u.h.textContent = pad(Math.floor(s % 86400 / 3600));
      u.m.textContent = pad(Math.floor(s % 3600 / 60));
      u.s.textContent = pad(s % 60);
      setTimeout(tick, 1000 - (Date.now() % 1000));
    }
    tick();
  })();
</script>
